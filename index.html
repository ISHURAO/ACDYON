const client = new AcdyonIngest({
  apiKey: process.env.ACDYON_KEY,
  behavior: 'stealth-dynamic'

  url: 'https://linkedin.com/jobs/search',
  schema: {

    salary: '.salary-range'

  options: {

    solveCaptchas: true,
    autoHealSelectors: true
  }
});
